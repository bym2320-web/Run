    return fallbackMs || 0;
  }

  // CSV 요약 컬럼(Elevation Gain / Ascent / 상승고도 등) 최우선 추출
  function extractElevationGainFromCsv(csvText) {
    if (!csvText) return 0;
    var lines = csvText.split(/\r?\n/).filter(function(l) { return l.trim() !== ""; });
    if (lines.length < 2) return 0;
    var hIdx = (lines[0].toLowerCase().indexOf("com.samsung.") === 0 && lines.length > 2) ? 1 : 0;
    var sep = (lines[hIdx].indexOf(";") !== -1 && lines[hIdx].indexOf(",") === -1) ? ";" : ",";
    var headers = lines[hIdx].split(sep).map(function(h) { return h.replace(/["']/g, "").trim().toLowerCase(); });
    var colIdx = -1;
    for (var i = 0; i < headers.length; i++) {
      var h = headers[i];
      if (/(elevation[_\s]*gain|altitude[_\s]*gain|total[_\s]*ascent|^ascent$|상승고도|고도\s*상승|누적\s*상승)/i.test(h)) {
        colIdx = i;
        break;
      }
    }
    if (colIdx === -1) return 0;
    var best = 0;
    for (var r = hIdx + 1; r < lines.length; r++) {
      var cols = lines[r].split(sep);
      if (cols.length <= colIdx) continue;
      var v = parseFloat(String(cols[colIdx]).replace(/[^0-9.-]/g, ""));
      if (!isNaN(v) && v > best && v <= 6000) best = v;
    }
    return Math.round(best);
  }

  // GPX/TCX 3단계 정밀 누적 상승고도 계산: ① 초반 GPS 튐 보정 ② 5포인트 이동평균 스무딩 ③ 1.5m 피크-밸리 히스테리시스
  function extractElevationGainFromXml(xmlText) {
    if (!xmlText) return 0;
    var re = /<(?:ele|AltitudeMeters)>\s*([0-9.-]+)\s*<\/(?:ele|AltitudeMeters)>/gi;
    var m, raw = [];
    while ((m = re.exec(xmlText)) !== null) {
      var alt = parseFloat(m[1]);
      if (!isNaN(alt) && alt > -500 && alt < 9000) raw.push(alt);
    }
    if (raw.length < 2) return 0;

    // ① 초반 안정화 구간(처음 8포인트) 3m 이상 GPS 초기 오차 스파이크 보정 + 단일 포인트 급튐 제거
    var cleaned = raw.slice();
    if (cleaned.length >= 12) {
      var stabSlice = cleaned.slice(5, Math.min(13, cleaned.length)).sort(function(a, b) { return a - b; });
      var stabMed = stabSlice[Math.floor(stabSlice.length / 2)];
      for (var si = 0; si < 8; si++) {
        if (Math.abs(cleaned[si] - stabMed) >= 3.0) cleaned[si] = stabMed;
      }
    }
    for (var ci = 1; ci < cleaned.length; ci++) {
      if (Math.abs(cleaned[ci] - cleaned[ci - 1]) >= 15.0) cleaned[ci] = cleaned[ci - 1];
    }

    // ② 5포인트 이동평균(Moving Average) 스무딩
    var halfWin = 2;
    var smoothed = [];
    for (var i = 0; i < cleaned.length; i++) {
      var sIdx = Math.max(0, i - halfWin);
      var eIdx = Math.min(cleaned.length - 1, i + halfWin);
      var sum = 0;
      for (var j = sIdx; j <= eIdx; j++) sum += cleaned[j];
      smoothed.push(sum / (eIdx - sIdx + 1));
    }

    // ③ 피크-밸리 히스테리시스(임계값 1.5m): 짧은 코스(5km 4m)는 살리고 장거리(25km 19m) 누적 노이즈는 억제
    var THRESHOLD = 1.5;
    var gain = 0;
    var dir = 0; // 0: init, 1: climbing, -1: descending
    var anchor = smoothed[0];
    var extreme = smoothed[0];
    for (var k = 1; k < smoothed.length; k++) {
      var cur = smoothed[k];
      if (dir === 0) {
        if (cur - anchor >= THRESHOLD) {
          dir = 1;
          extreme = cur;
        } else if (anchor - cur >= THRESHOLD) {
          dir = -1;
          extreme = cur;
        }
      } else if (dir === 1) {
        if (cur > extreme) {
          extreme = cur;
        } else if (extreme - cur >= THRESHOLD) {
          gain += (extreme - anchor);
          anchor = extreme;
          extreme = cur;
          dir = -1;
        }
      } else {
        if (cur < extreme) {
          extreme = cur;
        } else if (cur - extreme >= THRESHOLD) {
          anchor = extreme;
          extreme = cur;
          dir = 1;
        }
      }
    }
    if (dir === 1 && extreme - anchor >= THRESHOLD) {
      gain += (extreme - anchor);
    }
    return Math.round(gain);
  }

  // Read ALL sheets (if Google Sheet) or full CSV, scan EVERY row by real Date timestamp, return header + single newest row
  function extractNewestBodyRow(f) {
    var mime = f.getMimeType();
    var allLines = [];
    if (mime === MimeType.GOOGLE_SHEETS) {
      var sheets = SpreadsheetApp.openById(f.getId()).getSheets();
      for (var si = 0; si < sheets.length; si++) {
        var vals = sheets[si].getDataRange().getDisplayValues();
        if (!vals || vals.length === 0) continue;
        for (var ri = 0; ri < vals.length; ri++) {
          var rowStr = vals[ri].map(function(cell) {
            var s = String(cell || "").replace(/"/g, '""');
            return /[",\n]/.test(s) ? ('"' + s + '"') : s;
          }).join(",");
          if (rowStr.replace(/,/g, "").trim() !== "") allLines.push(rowStr);
        }
      }
    } else {
      allLines = f.getBlob().getDataAsString("UTF-8").split(/\r?\n/).filter(function(l) { return l.trim() !== ""; });
    }
    if (allLines.length === 0) return null;

    var headerIdx = 0;
    for (var hi = 0; hi < Math.min(5, allLines.length); hi++) {
      var low = allLines[hi].toLowerCase();
      if (low.indexOf("com.samsung.") === 0 && low.indexOf("start_time") === -1 && low.indexOf("create_time") === -1) continue;
      if (/(weight|body|fat|muscle|bmi|체중|몸무게|체지방|골격근|date|time|날짜)/i.test(low)) {
        headerIdx = hi;
        break;
      }
    }
    var header = allLines[headerIdx];
    var bestMs = 0, bestLine = "";
    var nowLimit = Date.now() + 86400000;
    for (var i = headerIdx + 1; i < allLines.length; i++) {
      var line = allLines[i];
      if (/^(date|start_time|measurement_time|날짜|측정일)/i.test(line.trim())) continue;
      var ms = extractDateMs(line, 0);
      if (ms > bestMs && ms <= nowLimit) {
        bestMs = ms;
        bestLine = line;
      }
    }
    if (!bestLine && allLines.length > headerIdx + 1) {
      bestLine = allLines[allLines.length - 1];
      bestMs = extractDateMs(f.getName(), f.getLastUpdated().getTime());
    }
    if (!bestLine) return null;
    return { maxMs: bestMs, csv: header + "\n" + bestLine };
  }

  // =========================================================================
  // 1) mode=today : 상위 5개 경량 .csv + 해당 5건에 매칭되는 .gpx/.tcx 고도(+m)만 정밀 보완
  // =========================================================================
  if (mode === "today" || mode === "running") {
    var limit = Math.min(10, parseInt(params.limit || "5", 10) || 5);
    var qCsv = "trashed = false and title contains '.csv' and (title contains 'RUNNING' or title contains 'running' or title contains 'Running' or title contains '달리기' or title contains '러닝')";
    var iter = DriveApp.searchFiles(qCsv);
    var list = [];
    while (iter.hasNext()) {
      var f = iter.next();
      var name = f.getName();
      if (excludeKw.test(name) || bodyKw.test(name)) continue;
      var modMs = f.getLastUpdated().getTime();
      list.push({ file: f, name: name, modMs: modMs, actMs: extractDateMs(name, modMs) });
    }
    list.sort(function(a, b) { return (b.actMs - a.actMs) || (b.modMs - a.modMs); });

    // Pick top N unique runs first
    var picked = [], seenMin = {};
    for (var i = 0; i < list.length && picked.length < limit; i++) {
      var it = list[i];
      if (it.file.getSize() > 1024 * 1024) continue;
      var k = Math.floor(it.actMs / 60000);
      if (seenMin[k]) continue;
      seenMin[k] = true;
      picked.push(it);
    }

    // Index matching .gpx/.tcx metadata ONLY for those top N runs (fast metadata check!)
    var pickedKeys = {};
    for (var pIdx = 0; pIdx < picked.length; pIdx++) {
      var pk = Math.floor(picked[pIdx].actMs / 60000);
      pickedKeys[pk] = true;
      pickedKeys[pk - 1] = true;
      pickedKeys[pk + 1] = true;
    }
    var elevFileByMin = {};
    if (picked.length > 0) {
      var qGeo = "trashed = false and (title contains '.tcx' or title contains '.gpx') and (title contains 'RUNNING' or title contains 'running' or title contains 'Running' or title contains '달리기' or title contains '러닝')";
      var gIter = DriveApp.searchFiles(qGeo);
      while (gIter.hasNext() && (Date.now() - startMs) < 12000) {
        var gf = gIter.next();
        var gname = gf.getName();
        if (excludeKw.test(gname) || bodyKw.test(gname)) continue;
        var gMs = extractDateMs(gname, gf.getLastUpdated().getTime());
        var gKey = Math.floor(gMs / 60000);
        if (pickedKeys[gKey] && !elevFileByMin[gKey]) {
          elevFileByMin[gKey] = gf;
        }
      }
    }

    var out = [];
    for (var oi = 0; oi < picked.length; oi++) {
      var pit = picked[oi];
      var csvText = pit.file.getBlob().getDataAsString("UTF-8");
      var elevGain = extractElevationGainFromCsv(csvText);
      if (elevGain <= 0) {
        var pk2 = Math.floor(pit.actMs / 60000);
        var matchedGeo = elevFileByMin[pk2] || elevFileByMin[pk2 - 1] || elevFileByMin[pk2 + 1] || null;
        if (matchedGeo && matchedGeo.getSize() <= 5 * 1024 * 1024) {
          elevGain = extractElevationGainFromXml(matchedGeo.getBlob().getDataAsString("UTF-8"));
        }
      }
      out.push({
        fileName: pit.name,
        actualDateMs: pit.actMs,
        date: Utilities.formatDate(new Date(pit.actMs || pit.modMs), "Asia/Seoul", "yyyy-MM-dd"),
        modifiedTime: new Date(pit.modMs).toISOString(),
        elevGain: elevGain,
        content: csvText
      });
    }
    return ContentService.createTextOutput(JSON.stringify({ mode: "today", files: out })).setMimeType(ContentService.MimeType.JSON);
  }

  // =========================================================================
  // 2) mode=weight : 모든 체중/체성분 파일 전수 날짜 비교 -> 가장 최신 1건 정확 반환
  // =========================================================================
  if (mode === "weight" || mode === "body") {
    var candidates = [], seen = {};
    function addW(f) {
      var id = f.getId();
      if (seen[id]) return;
      seen[id] = true;
      var name = f.getName();
      if (excludeKw.test(name)) return;
      var mime = f.getMimeType();
      if (mime !== MimeType.GOOGLE_SHEETS && !/\.(csv|txt|json)$/i.test(name)) return;
      var modMs = f.getLastUpdated().getTime();
      candidates.push({ file: f, name: name, modMs: modMs, actMs: extractDateMs(name, modMs) });
    }
    var cachedFolderId = props.getProperty("RG_WEIGHT_FOLDER_ID");
    if (cachedFolderId) {
      try {
        var cf = DriveApp.getFolderById(cachedFolderId);
        var cfi = cf.getFiles();
        while (cfi.hasNext()) addW(cfi.next());
      } catch (_) { props.deleteProperty("RG_WEIGHT_FOLDER_ID"); }
    }
    var fIter = DriveApp.searchFolders("trashed = false and (title contains 'Weight' or title contains 'weight' or title contains 'Body' or title contains 'body' or title contains '체중' or title contains '체성분' or title contains '무게')");
    while (fIter.hasNext() && (Date.now() - startMs) < 12000) {
      var wf = fIter.next();
      if (!cachedFolderId) { props.setProperty("RG_WEIGHT_FOLDER_ID", wf.getId()); cachedFolderId = wf.getId(); }
      var wfi = wf.getFiles();
      while (wfi.hasNext()) addW(wfi.next());
    }
    var bIter = DriveApp.searchFiles("trashed = false and (title contains 'Weight' or title contains 'weight' or title contains 'WEIGHT' or title contains 'Body' or title contains 'body' or title contains '체중' or title contains '체성분' or title contains '무게' or title contains 'InBody')");
    while (bIter.hasNext() && (Date.now() - startMs) < 20000) addW(bIter.next());

    candidates.sort(function(a, b) { return (b.actMs - a.actMs) || (b.modMs - a.modMs); });
    var bestObj = null;
    // Check ALL weight/body files (no 5-file cap!) so the true newest date (2026-11) always wins
    for (var wi = 0; wi < candidates.length; wi++) {
      if ((Date.now() - startMs) > 55000) break;
      var item = candidates[wi];
      if (item.file.getMimeType() !== MimeType.GOOGLE_SHEETS && item.file.getSize() > 3 * 1024 * 1024) continue;
      var parsed = extractNewestBodyRow(item.file);
      if (!parsed || !parsed.maxMs) continue;
      if (!bestObj || parsed.maxMs > bestObj.contentMaxDateMs) {
        bestObj = {
          fileName: (item.file.getMimeType() === MimeType.GOOGLE_SHEETS ? item.name + ".csv" : item.name),
          actualDateMs: parsed.maxMs,
          contentMaxDateMs: parsed.maxMs,
          date: Utilities.formatDate(new Date(parsed.maxMs), "Asia/Seoul", "yyyy-MM-dd"),
          modifiedTime: new Date(item.modMs).toISOString(),
          content: parsed.csv
        };
      }
    }
    var arr = bestObj ? [bestObj] : [];
    return ContentService.createTextOutput(JSON.stringify({ mode: "weight", totalScanned: candidates.length, files: arr, bodyFiles: arr })).setMimeType(ContentService.MimeType.JSON);
  }

  // =========================================================================
  // 3) mode=pb : PropertiesService 마지막 스캔 시각 기반 증분 조회
  // =========================================================================
  if (forceRescan) {
    props.deleteProperty("RG_PB_LAST_SCAN_ISO");
    props.deleteProperty("RG_PB_BEST_JSON");
  }
  var lastScanIso = forceRescan ? "" : (props.getProperty("RG_PB_LAST_SCAN_ISO") || "");
  var cachedRaw = forceRescan ? "" : (props.getProperty("RG_PB_BEST_JSON") || "");
  var cachedFiles = [];
  try { if (cachedRaw) cachedFiles = JSON.parse(cachedRaw); } catch (_) { cachedFiles = []; }

  var qPb = "trashed = false and title contains '.csv' and (title contains 'RUNNING' or title contains 'running' or title contains 'Running' or title contains '달리기' or title contains '러닝')";
  if (lastScanIso) qPb += " and modifiedDate > '" + lastScanIso + "'";

  var pIter = DriveApp.searchFiles(qPb);
  var newList = [];
  while (pIter.hasNext() && (Date.now() - startMs) < 25000) {
    var pf = pIter.next();
    var pname = pf.getName();
    if (excludeKw.test(pname) || bodyKw.test(pname)) continue;
    var pmod = pf.getLastUpdated().getTime();
    newList.push({ file: pf, name: pname, modMs: pmod, actMs: extractDateMs(pname, pmod) });
  }
  newList.sort(function(a, b) { return b.actMs - a.actMs; });

  var newOut = [], seenPbMin = {}, totalB = 0;
  for (var pi = 0; pi < newList.length && newOut.length < 150; pi++) {
    if ((Date.now() - startMs) > 60000) break;
    var pit = newList[pi];
    if (pit.file.getSize() > 500 * 1024) continue;
    var pk = Math.floor(pit.actMs / 60000);
    if (seenPbMin[pk]) continue;
    seenPbMin[pk] = true;
    var ptxt = pit.file.getBlob().getDataAsString("UTF-8");
    totalB += ptxt.length;
    if (totalB > 8 * 1024 * 1024) break;
    newOut.push({
      fileName: pit.name,
      actualDateMs: pit.actMs,
      date: Utilities.formatDate(new Date(pit.actMs || pit.modMs), "Asia/Seoul", "yyyy-MM-dd"),
      modifiedTime: new Date(pit.modMs).toISOString(),
      content: ptxt
    });
  }

  props.setProperty("RG_PB_LAST_SCAN_ISO", new Date().toISOString().slice(0, 19));
  return ContentService.createTextOutput(JSON.stringify({
    mode: "pb",
    incremental: !!lastScanIso,
    newlyScanned: newOut.length,
    files: newOut,
    cachedWorkouts: cachedFiles
  })).setMimeType(ContentService.MimeType.JSON);
}
